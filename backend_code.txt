import { Injectable } from '@nestjs/common';
import { InjectRepository } from '@nestjs/typeorm';
import { Repository } from 'typeorm';
import { AuditLog } from './auditlog.entity';

@Injectable()
export class AuditLogService {
  constructor(
    @InjectRepository(AuditLog)
    private readonly auditLogRepository: Repository<AuditLog>,
  ) {}

  async recordAction(data: {
    userId: number;
    actionType: string;
    entityType: string;
    entityId: number;
    details?: string;
  }): Promise<AuditLog> {
    const auditLog = this.auditLogRepository.create(data);

    return this.auditLogRepository.save(auditLog);
  }

  async getAllLogs(): Promise<AuditLog[]> {
    return this.auditLogRepository.find({
      order: { createdAt: 'DESC' },
    });
  }

  async getLogsForRecord(
    entityType: string,
    entityId: number,
  ): Promise<AuditLog[]> {
    return this.auditLogRepository.find({
      where: { entityType, entityId } as any,
      order: { createdAt: 'DESC' },
    });
  }
}

import {CanActivate, ExecutionContext, Injectable, UnauthorizedException,} from '@nestjs/common';
import { JwtService } from '@nestjs/jwt';
import { UserRole } from '../../users/user-role.enum';
import { AuthenticatedRequest } from '../interfaces/authenticated-request.interface';
import { AuthTokenPayload } from '../interfaces/auth-token-payload.interface';

@Injectable()
export class JwtAuthGuard implements CanActivate {
  constructor(private readonly jwtService: JwtService) {}

  async canActivate(context: ExecutionContext): Promise<boolean> {
    const request =
      context.switchToHttp().getRequest<AuthenticatedRequest>();
    const authorization = request.headers.authorization;
    const parts = authorization?.split(' ') ?? [];
    const [scheme, token] = parts;

    if (parts.length !== 2 || scheme !== 'Bearer' || !token) {
      throw new UnauthorizedException('A bearer access token is required');
    }

    let payload: AuthTokenPayload;
    try {
      payload = await this.jwtService.verifyAsync<AuthTokenPayload>(token);
    } catch (error) {
      if (
        error instanceof Error &&
        ['JsonWebTokenError', 'TokenExpiredError', 'NotBeforeError'].includes(
          error.name,
        )
      ) {
        throw new UnauthorizedException('The access token is invalid or expired');
      }
      throw error;
    }

    if (
      typeof payload !== 'object' ||
      payload === null ||
      typeof payload.sub !== 'string' ||
      typeof payload.email !== 'string' ||
      (payload.role !== null &&
        !Object.values(UserRole).includes(payload.role))
    ) {
      throw new UnauthorizedException('The access token is invalid');
    }

    request.user = payload;
    return true;
  }
}

import {
  CanActivate,
  ExecutionContext,
  Injectable,
  UnauthorizedException,
} from '@nestjs/common';
import { Reflector } from '@nestjs/core';
import { UserRole } from '../../users/user-role.enum';
import { ROLES_KEY } from '../decorators/roles.decorator';
import { AuthenticatedRequest } from '../interfaces/authenticated-request.interface';

@Injectable()
export class RolesGuard implements CanActivate {
  constructor(private readonly reflector: Reflector) {}

  canActivate(context: ExecutionContext): boolean {
    const requiredRoles = this.reflector.getAllAndOverride<UserRole[]>(
      ROLES_KEY,
      [context.getHandler(), context.getClass()],
    );
    if (!requiredRoles?.length) {
      return true;
    }

    const { user } =
      context.switchToHttp().getRequest<AuthenticatedRequest>();
    if (!user) {
      throw new UnauthorizedException('Authentication is required');
    }

    return user.role !== null && requiredRoles.includes(user.role);
  }
}

import { Module } from '@nestjs/common';
import { ConfigModule, ConfigService } from '@nestjs/config';
import { JwtModule } from '@nestjs/jwt';
import { JwtAuthGuard } from './guards/jwt-auth.guard';
import { RolesGuard } from './guards/roles.guard';

@Module({
  imports: [
    ConfigModule,
    JwtModule.registerAsync({
      imports: [ConfigModule],
      inject: [ConfigService],
      useFactory: (configService: ConfigService) => {
        const secret = configService.get<string>('JWT_SECRET');
        if (!secret || secret.length < 16) {
          throw new Error('JWT_SECRET must contain at least 16 characters');
        }
        return {
          secret,
          signOptions: { expiresIn: '1h' },
        };
      },
    }),
  ],
  providers: [JwtAuthGuard, RolesGuard],
  exports: [JwtModule, JwtAuthGuard, RolesGuard],
})
export class AuthSecurityModule {}

import { Injectable, UnauthorizedException } from '@nestjs/common';
import * as bcrypt from 'bcrypt';
import { JwtService } from '@nestjs/jwt';
import { InjectRepository } from '@nestjs/typeorm';
import { Repository } from 'typeorm';
import { User } from '../users/user.entity';
import { LoginDto } from './dto/login.dto';
import { AuthTokenPayload } from './interfaces/auth-token-payload.interface';
import { ForgotPasswordDto } from './dto/forgot-password.dto';

const BCRYPT_ROUNDS = 12;

@Injectable()
export class AuthService {
  constructor(
    @InjectRepository(User)
    private readonly usersRepository: Repository<User>,
    private readonly jwtService: JwtService,
  ) {}

  async login(loginDto: LoginDto) {
    const email = loginDto.email.trim().toLowerCase();

    const user = await this.usersRepository
      .createQueryBuilder('user')
      .addSelect('user.passwordHash')
      .where('user.email = :email', { email })
      .getOne();

    if (!user) {
      await bcrypt.hash(loginDto.password, BCRYPT_ROUNDS);
      throw new UnauthorizedException('Invalid email or password');
    }

    const passwordMatches = await this.verifyPassword(loginDto.password, user.passwordHash,);

    if (!passwordMatches) {
      throw new UnauthorizedException('Invalid email or password');
    }

    return this.loginUser(user);
  }

  async forgotPassword(_forgotPasswordDto: ForgotPasswordDto) {
    // Keep the response independent of whether the email belongs to an account.
    // Password reset delivery can be connected here when an email provider is configured.
    return {
      message:
        'If an account exists for this email, password reset instructions will be sent.',
    };
  }

  private async loginUser(user: User){
    const payload: AuthTokenPayload = {
      sub: user.id,
      email: user.email,
      role: user.role,
    }

    const accessToken = await this.jwtService.signAsync(payload);

    return {accessToken}
  }

  private async verifyPassword(password: string, passwordHash: string,): Promise<boolean> {
    return bcrypt.compare(password, passwordHash);
  }
}

import { Injectable } from '@nestjs/common';
import { DataSource } from 'typeorm';

@Injectable()
export class DashboardService {
  constructor(private readonly dataSource: DataSource) {}

  // Summary cards
  async getSummary(departmentId?: number) {
    const params: any[] = [];
    let departmentCondition = '';

    if (departmentId !== undefined) {
      departmentCondition = 'WHERE department_id = ?';
      params.push(departmentId);
    }

    const [result] = await this.dataSource.query(
      `SELECT
         COUNT(*) AS totalRequests,
         COALESCE(SUM(status = 'PENDING'), 0) AS pendingApprovals,
         COALESCE(SUM(status = 'APPROVED'), 0) AS approvedRequests,
         COALESCE(SUM(status = 'REJECTED'), 0) AS rejectedRequests
       FROM resource_requests
       ${departmentCondition}`,
      params,
    );

    const allocationParams: any[] = [];
    let allocationCondition = '';

    if (departmentId !== undefined) {
      allocationCondition = 'WHERE department_id = ?';
      allocationParams.push(departmentId);
    }

    const [allocationResult] = await this.dataSource.query(
      `SELECT COUNT(*) AS allocatedResources
       FROM resource_allocations
       ${allocationCondition}`,
      allocationParams,
    );

    return {
      totalRequests: Number(result.totalRequests),
      pendingApprovals: Number(result.pendingApprovals),
      approvedRequests: Number(result.approvedRequests),
      rejectedRequests: Number(result.rejectedRequests),
      allocatedResources: Number(
        allocationResult.allocatedResources,
      ),
    };
  }

  // Recent audit activity
  async getRecentActivity(limit = 10) {
    const safeLimit = Math.min(Math.max(limit, 1), 50);

    return this.dataSource.query(
      `SELECT *
       FROM audit_logs
       ORDER BY created_at DESC
       LIMIT ${safeLimit}`,
    );
  }

  // Spending summary grouped by department
  async getDepartmentSpending() {
    return this.dataSource.query(
      `SELECT
         d.id AS departmentId,
         d.name AS departmentName,
         COALESCE(SUM(
           CASE
             WHEN r.status = 'APPROVED' THEN r.total_cost
             ELSE 0
           END
         ), 0) AS totalSpending
       FROM departments d
       LEFT JOIN resource_requests r
         ON r.department_id = d.id
       GROUP BY d.id, d.name
       ORDER BY totalSpending DESC`,
    );
  }

  // Complete dashboard data
  async getDashboard(departmentId?: number) {
    return {
      summary: await this.getSummary(departmentId),
      recentActivity: await this.getRecentActivity(),
      departmentSpending: await this.getDepartmentSpending(),
    };
  }
}

import {
  ForbiddenException,
  Injectable,
  NotFoundException,
} from '@nestjs/common';
import { InjectRepository } from '@nestjs/typeorm';
import { Repository } from 'typeorm';
import { Department } from './entities/department.entity';
import { CreateDepartmentDto } from './dto/create-department.dto';
import { UpdateDepartmentDto } from './dto/update-departments.dto';
import { User } from '../users/user.entity';
import { Team } from '../teams/entities/team.entity';
import { Project } from '../projects/entities/project.entity';
import { ResourceRequest } from '../resourcerequest/entities/resourcerequest.entity';
import { AuthTokenPayload } from '../auth/interfaces/auth-token-payload.interface';
import { UserRole } from '../users/user-role.enum';

@Injectable()
export class DepartmentsService {
  constructor(
    @InjectRepository(Department)
    private readonly departmentRepository: Repository<Department>,
    @InjectRepository(User)
    private readonly userRepository: Repository<User>,
    @InjectRepository(Team)
    private readonly teamRepository: Repository<Team>,
    @InjectRepository(Project)
    private readonly projectRepository: Repository<Project>,
    @InjectRepository(ResourceRequest)
    private readonly resourceRequestRepository: Repository<ResourceRequest>,
  ) {}

  async create(createDepartmentDto: CreateDepartmentDto) {
    const department = this.departmentRepository.create(createDepartmentDto);
    return this.departmentRepository.save(department);
  }

  async findAll() {
    return this.departmentRepository.find();
  }

  async findOne(id: number) {
    const department = await this.departmentRepository.findOne({ where: { id } });

    if (!department) {
      throw new NotFoundException(`Department with id ${id} not found`);
    }

    return department;
  }

  async findUsers(id: number) {
    await this.findOne(id);
    return this.userRepository.find({ where: { department: { id } } });
  }

  async findTeams(id: number) {
    await this.findOne(id);
    return this.teamRepository.find({ where: { department: { id } } });
  }

  async findProjects(id: number) {
    await this.findOne(id);
    return this.projectRepository.find({
      where: { teams: { department: { id } } },
      relations: { teams: true },
    });
  }

  async findResourceRequests(id: number, user: AuthTokenPayload) {
    if (user.role !== UserRole.DEPARTMENT_HEAD) {
      throw new ForbiddenException('Only department heads can view department requests');
    }

    const account = await this.userRepository.findOne({
      where: { id: user.sub },
      relations: { department: true },
    });
    if (!account?.department) {
      throw new ForbiddenException('Your account is not assigned to a department');
    }
    if (account.department.id !== id) {
      throw new NotFoundException(`Department with id ${id} not found`);
    }

    return this.resourceRequestRepository.find({
      where: { departmentId: String(account.department.id) },
      order: { createdAt: 'DESC' },
    });
  }

  async findTeamProjects(departmentId: number, teamId: number) {
    await this.findOne(departmentId);
    const team = await this.teamRepository.findOne({
      where: { id: teamId, department: { id: departmentId } },
    });

    if (!team) {
      throw new NotFoundException(
        `Team with id ${teamId} not found in department ${departmentId}`,
      );
    }

    return this.projectRepository.find({
      where: { teams: { id: teamId } },
      relations: { teams: true },
    });
  }

  async update(id: number, updateDepartmentDto: UpdateDepartmentDto) {
    const department = await this.findOne(id);
    Object.assign(department, updateDepartmentDto);
    return this.departmentRepository.save(department);
  }

  async remove(id: number) {
    const department = await this.findOne(id);
    await this.departmentRepository.remove(department);
    return department;
  }
}

import { BadRequestException, ForbiddenException, Injectable, NotFoundException, UnprocessableEntityException } from '@nestjs/common';
import { InjectRepository } from '@nestjs/typeorm';
import { In, Repository } from 'typeorm';
import { Team } from '../teams/entities/team.entity';
import { User } from '../users/user.entity';
import { UserRole } from '../users/user-role.enum';
import { AuthTokenPayload } from '../auth/interfaces/auth-token-payload.interface';
import { CreateProjectDto } from './dto/create-project.dto';
import { Project } from './entities/project.entity';
import { UpdateProjectDto } from './dto/update-project.dto';

@Injectable()
export class ProjectsService {
  constructor(
    @InjectRepository(Project)
    private readonly projectRepository: Repository<Project>,
    @InjectRepository(Team)
    private readonly teamRepository: Repository<Team>,
    @InjectRepository(User)
    private readonly users: Repository<User>,
  ) {}

  async create(createProjectDto: CreateProjectDto, user: AuthTokenPayload) {
    const departmentId = await this.getDepartmentHeadDepartmentId(user);
    const { teamIds = [], ...projectDetails } = createProjectDto;
    if (teamIds.length === 0) {
      throw new UnprocessableEntityException('A project must be assigned to a team in your department');
    }
    const teams = await this.findTeams(teamIds);
    this.assertTeamsInDepartment(teams, departmentId);
    const project = this.projectRepository.create({
      ...projectDetails,
      teams,
    });
    return this.projectRepository.save(project);
  }


  async findAll(user: AuthTokenPayload) {
    if (user.role === UserRole.MANAGING_DIRECTOR) {
      return this.projectRepository.find({
        relations: { teams: { department: true } },
      });
    }
    const departmentId = await this.getDepartmentHeadDepartmentId(user);
    return this.projectRepository.find({
      where: { teams: { department: { id: departmentId } } },
      relations: { teams: { department: true } },
    });
  }

  async findOne(id: number, user: AuthTokenPayload) {
    const project = await this.projectRepository.findOne({
      where: { id },
      relations: { teams: { department: true } },
    });
    if (!project) {
      throw new NotFoundException(`Project with id ${id} not found`);
    }
    if (user.role !== UserRole.MANAGING_DIRECTOR) {
      const departmentId = await this.getDepartmentHeadDepartmentId(user);
      if (!project.teams.some((team) => team.department?.id === departmentId)) {
        throw new NotFoundException(`Project with id ${id} not found`);
      }
    }
    return project;
  }

  async assignTeams(id: number, teamIds: number[], user: AuthTokenPayload) {
    const departmentId = await this.getDepartmentHeadDepartmentId(user);
    const project = await this.findOne(id, user);
    const teams = await this.findTeams(teamIds);
    this.assertTeamsInDepartment(teams, departmentId);
    if (teams.length === 0) {
      throw new UnprocessableEntityException('A project must remain assigned to a team in your department');
    }
    project.teams = teams;
    return this.projectRepository.save(project);
  }

  async update(id: number, updateProjectDto: UpdateProjectDto, user: AuthTokenPayload) {
    if (updateProjectDto.name === undefined && updateProjectDto.description === undefined) {
      throw new BadRequestException('Provide a project name or description to update');
    }
    const project = await this.findOne(id, user);
    Object.assign(project, updateProjectDto);
    return this.projectRepository.save(project);
  }

  async findTeamMembers(id: number, user: AuthTokenPayload) {
    const project = await this.findOne(id, user);
    const teamIds = project.teams.map((team) => team.id);
    if (teamIds.length === 0) return [];

    const members = await this.users.find({
      where: { team: { id: In(teamIds) } },
      relations: { team: true },
      select: { id: true, fullName: true, team: { id: true, name: true } },
      order: { fullName: 'ASC' },
    });
    return members.map((member) => ({
      id: member.id,
      fullName: member.fullName,
      teamId: member.team?.id,
      teamName: member.team?.name,
    }));
  }

  async remove(id: number, user: AuthTokenPayload) {
    const project = await this.findOne(id, user);
    await this.projectRepository.remove(project);
    return { message: `Project with id ${id} deleted successfully` };
  }

  private async getDepartmentHeadDepartmentId(user: AuthTokenPayload): Promise<number> {
    if (user.role !== UserRole.DEPARTMENT_HEAD) {
      throw new ForbiddenException('Only department heads can create or manage projects');
    }
    const account = await this.users.findOne({
      where: { id: user.sub },
      relations: { department: true },
    });
    if (!account?.department) {
      throw new ForbiddenException('Your account is not assigned to a department');
    }
    return account.department.id;
  }

  private assertTeamsInDepartment(teams: Team[], departmentId: number) {
    const invalidTeams = teams.filter((team) => team.department?.id !== departmentId);
    if (invalidTeams.length > 0) {
      throw new ForbiddenException('Projects can only be assigned to teams in your department');
    }
  }

  private async findTeams(teamIds: number[]) {
    if (teamIds.length === 0) return [];

    const teams = await this.teamRepository.find({ where: { id: In(teamIds) } });
    if (teams.length !== teamIds.length) {
      const foundIds = new Set(teams.map((team) => team.id));
      const missingIds = teamIds.filter((id) => !foundIds.has(id));
      throw new UnprocessableEntityException(
        `Team ids not found: ${missingIds.join(', ')}`,
      );
    }
    return teams;
  }
}

import { Injectable } from '@nestjs/common';
import { DataSource } from 'typeorm';
import { ReportFilterDto } from './dto/report-filter.dto';

@Injectable()
export class ReportsService {
  constructor(private readonly dataSource: DataSource) {}

  // 1. Historical approval decisions
  async getApprovalHistory(filter: ReportFilterDto) {
    const conditions: string[] = [];
    const params: any[] = [];

    if (filter.departmentId) {
      params.push(filter.departmentId);
      conditions.push(`r.department_id = ?`);
    }

    if (filter.startDate) {
      params.push(filter.startDate);
      conditions.push(`a.created_at >= ?`);
    }

    if (filter.endDate) {
      params.push(filter.endDate);
      conditions.push(`a.created_at < DATE_ADD(?, INTERVAL 1 DAY)`);
    }

    const where = conditions.length
      ? `WHERE ${conditions.join(' AND ')}`
      : '';

    return this.dataSource.query(
      `SELECT a.*, r.department_id
       FROM approvals a
       JOIN resource_requests r ON r.id = a.request_id
       ${where}
       ORDER BY a.created_at DESC`,
      params,
    );
  }

  // 2. Departmental spending patterns
  async getDepartmentalSpending(filter: ReportFilterDto) {
    const conditions: string[] = [
      `r.status = 'APPROVED'`,
    ];
    const params: any[] = [];

    if (filter.departmentId) {
      params.push(filter.departmentId);
      conditions.push(`r.department_id = ?`);
    }

    if (filter.startDate) {
      params.push(filter.startDate);
      conditions.push(`r.created_at >= ?`);
    }

    if (filter.endDate) {
      params.push(filter.endDate);
      conditions.push(
        `r.created_at < DATE_ADD(?, INTERVAL 1 DAY)`,
      );
    }

    return this.dataSource.query(
      `SELECT d.id AS departmentId,
              d.name AS departmentName,
              COALESCE(SUM(r.total_cost), 0) AS totalSpending
       FROM departments d
       LEFT JOIN resource_requests r
         ON r.department_id = d.id
         AND ${conditions.join(' AND ')}
       GROUP BY d.id, d.name
       ORDER BY totalSpending DESC`,
      params,
    );
  }

  // 3. Resource allocation history
  async getAllocationHistory(filter: ReportFilterDto) {
    const conditions: string[] = [];
    const params: any[] = [];

    if (filter.departmentId) {
      params.push(filter.departmentId);
      conditions.push(`r.department_id = ?`);
    }

    if (filter.startDate) {
      params.push(filter.startDate);
      conditions.push(`ra.created_at >= ?`);
    }

    if (filter.endDate) {
      params.push(filter.endDate);
      conditions.push(
        `ra.created_at < DATE_ADD(?, INTERVAL 1 DAY)`,
      );
    }

    const where = conditions.length
      ? `WHERE ${conditions.join(' AND ')}`
      : '';

    return this.dataSource.query(
      `SELECT ra.*, r.department_id
       FROM resource_allocations ra
       JOIN resource_requests r ON r.id = ra.request_id
       ${where}
       ORDER BY ra.created_at DESC`,
      params,
    );
  }

  async getDepartmentReport(
    departmentId: number,
    filter: ReportFilterDto,
  ) {
    return {
      departmentId,
      approvals: await this.getApprovalHistory({
        ...filter,
        departmentId,
      }),
      spending: await this.getDepartmentalSpending({
        ...filter,
        departmentId,
      }),
      allocations: await this.getAllocationHistory({
        ...filter,
        departmentId,
      }),
    };
  }

  // 5. Organization-wide report
  async getOrganizationReport(filter: ReportFilterDto) {
    const organizationFilter = {
      startDate: filter.startDate,
      endDate: filter.endDate,
    };

    return {
      approvals: await this.getApprovalHistory(organizationFilter),
      spending: await this.getDepartmentalSpending(organizationFilter),
      allocations: await this.getAllocationHistory(organizationFilter),

  }
}

import {
  BadRequestException,
  ForbiddenException,
  Injectable,
  NotFoundException,
} from '@nestjs/common';
import { InjectRepository } from '@nestjs/typeorm';
import { FindOptionsWhere, Not, Repository } from 'typeorm';
import {
  RequestStatus,
  ResourceRequest,
} from './entities/resourcerequest.entity';
import { RequestAuditLog } from './entities/request-audit-log.entity';
import { CreateResourceRequestDto } from './dto/create-resourcerequest.dto';
import { UpdateResourceRequestDto } from './dto/update-resourcerequest.dto';
import { QueryResourceRequestDto } from './dto/query-resourcerequest.dto';
import { AuthUser, UserRole } from './roles';

// The only statuses a line manager is allowed to set.
const REVIEW_DECISIONS = [
  RequestStatus.APPROVED,
  RequestStatus.REJECTED,
  RequestStatus.QUERIED,
];

@Injectable()
export class ResourcerequestService {
  constructor(
    @InjectRepository(ResourceRequest)
    private readonly requests: Repository<ResourceRequest>,
       @InjectRepository(RequestAuditLog)
    private readonly audit: Repository<RequestAuditLog>,
  ) {}

  // ---------------------------------------------------------------
  // Create (team lead). Always starts as DRAFT.
  // ---------------------------------------------------------------
  async create(dto: CreateResourceRequestDto, user: AuthUser) {
    if (user.role !== UserRole.TEAM_LEAD) {
      throw new ForbiddenException('Only team leads can create requests');
    }
    // TODO: check teamId, departmentId and projectId against the
    // organization module (they exist, and the team belongs to the department).
    const request = this.requests.create({
      ...dto,
      requestedBy: user.id, // from the logged-in user, never from the body
      status: RequestStatus.DRAFT,
    });
    const saved = await this.requests.save(request);
    await this.log(saved.id, 'CREATED', user.id);
    return saved;
  }

  // ---------------------------------------------------------------
  // Read: one list endpoint, results depend on who is asking
  //   team lead:    the requests they created
  //   line manager: submitted requests routed to them (drafts hidden)
  //   dept head / managing director: everything
  // ---------------------------------------------------------------
  async findAll(query: QueryResourceRequestDto, user: AuthUser) {
    const where: FindOptionsWhere<ResourceRequest> = {};
    if (query.status) where.status = query.status;
    if (query.teamId) where.teamId = query.teamId;

    if (user.role === UserRole.TEAM_LEAD) {
      where.requestedBy = user.id;
    } else if (user.role === UserRole.LINE_MANAGER) {
      where.lineManagerId = user.id;
      if (query.status === RequestStatus.DRAFT) return [];
      if (!query.status) where.status = Not(RequestStatus.DRAFT);
    }
    // TODO: scope DEPARTMENT_HEAD to their own department once the
    // organization module can tell us which department they head.

    return this.requests.find({ where, order: { createdAt: 'DESC' } });
  }

  // One request. With includeAudit, the audit trail is attached too.
  async findOne(id: string, user: AuthUser, includeAudit = false) {
    const request = await this.getOrFail(id);
    this.assertCanView(request, user);
    if (!includeAudit) return request;

    const auditTrail = await this.audit.find({
      where: { requestId: id },
      order: { createdAt: 'ASC' },
    });
    return { ...request, auditTrail };
  }

  // ---------------------------------------------------------------
  // Write: one update endpoint
  //   team lead: edit details, change team/department, status SUBMITTED
  //   line manager: status APPROVED / REJECTED / QUERIED (with note)
  // ---------------------------------------------------------------
  async update(id: string, dto: UpdateResourceRequestDto, user: AuthUser) {
    const request = await this.getOrFail(id);
    const { status, note, ...fields } = dto;

    // Keep only the fields that were actually sent.
    const changes: Record<string, unknown> = Object.fromEntries(
      Object.entries(fields).filter(([, value]) => value !== undefined),
    );
    const changedKeys = Object.keys(changes);

    if (status === undefined && changedKeys.length === 0) {
      throw new BadRequestException('Nothing to update');
    }

    // A review decision (line manager). It can't be mixed with edits.
    if (status !== undefined && status !== RequestStatus.SUBMITTED) {
      if (changedKeys.length > 0) {
        throw new BadRequestException(
          'A review decision cannot be combined with edits',
        );
      }
      return this.review(request, status, note, user);
    }

    // Everything below is the team lead's: edits and submitting.
    this.assertOwner(request, user);
    this.assertEditable(request);
    // TODO: when teamId or departmentId change, check them against the
    // organization module (team exists, belongs to the department).
    Object.assign(request, changes);

    const submitting = status === RequestStatus.SUBMITTED;
    if (submitting) {
      // SRS rule: a request is only valid if it comes from a team that has
      // a department, and it must be tied to a project and a line manager.
      const missing: string[] = [];
      if (!request.teamId) missing.push('team');
      if (!request.departmentId) missing.push('department');
      if (!request.projectId) missing.push('project');
      if (!request.lineManagerId) missing.push('line manager');
      if (missing.length > 0) {
        throw new BadRequestException(
          `Cannot submit: missing ${missing.join(', ')}`,
        );
      }
      request.status = RequestStatus.SUBMITTED;
      request.submittedAt = new Date();
    }

    const saved = await this.requests.save(request);

    // One audit entry per kind of change.
    if (changedKeys.includes('teamId')) {
      await this.log(id, 'TEAM_CHANGED', user.id, request.teamId);
    }
    if (changedKeys.includes('departmentId')) {
      await this.log(id, 'DEPARTMENT_CHANGED', user.id, request.departmentId);
    }
    if (changedKeys.some((key) => key !== 'teamId' && key !== 'departmentId')) {
      await this.log(id, 'UPDATED', user.id);
    }
    if (submitting) {
      await this.log(id, 'SUBMITTED', user.id);
      // TODO: notification component:
      //  - notify the line manager that a request arrived
      //  - notify the team lead that the request was submitted
    }
    return saved;
  }

  // Line manager approves, rejects, or queries a SUBMITTED request.
  private async review(
    request: ResourceRequest,
    status: RequestStatus,
    note: string | undefined,
    user: AuthUser,
  ) {
    if (user.role !== UserRole.LINE_MANAGER) {
      throw new ForbiddenException(
        'Only line managers can approve, reject, or query requests',
      );
    }
    if (request.lineManagerId !== user.id) {
      throw new ForbiddenException('This request is routed to another manager');
    }
    if (request.status !== RequestStatus.SUBMITTED) {
      throw new BadRequestException(
        `Only SUBMITTED requests can be reviewed (current: ${request.status})`,
      );
    }
    if (!REVIEW_DECISIONS.includes(status)) {
      throw new BadRequestException(
        'Status must be SUBMITTED (team lead) or APPROVED, REJECTED, QUERIED (line manager)',
      );
    }
    if (status !== RequestStatus.APPROVED && !note?.trim()) {
      throw new BadRequestException(
        'A note is required when rejecting or querying a request',
      );
    }

    request.status = status;
    request.reviewNote = note ?? null;
    request.reviewedBy = user.id;
    request.reviewedAt = new Date();
    const saved = await this.requests.save(request);
    await this.log(request.id, status, user.id, note);

    // TODO: notification component: tell the team lead the outcome.
    return saved;
  }

  // ---------------------------------------------------------------
  // Private helpers
  // ---------------------------------------------------------------
  private async getOrFail(id: string) {
    const request = await this.requests.findOne({ where: { id } });
    if (!request) throw new NotFoundException(`Request ${id} not found`);
    return request;
  }

  private assertOwner(request: ResourceRequest, user: AuthUser) {
    if (request.requestedBy !== user.id) {
      throw new ForbiddenException('You do not own this request');
    }
  }

  // Requests can only be changed before review, or after a query.
  private assertEditable(request: ResourceRequest) {
    const editable = [RequestStatus.DRAFT, RequestStatus.QUERIED];
    if (!editable.includes(request.status)) {
      throw new BadRequestException(
        `A ${request.status} request can no longer be changed`,
      );
    }
  }

  private assertCanView(request: ResourceRequest, user: AuthUser) {
    const isOwner = request.requestedBy === user.id;
    const isAssignedManager = request.lineManagerId === user.id;
    const isOversight =
      user.role === UserRole.DEPARTMENT_HEAD ||
      user.role === UserRole.MANAGING_DIRECTOR;
    // TODO: limit DEPARTMENT_HEAD to requests in their own department.
    if (!isOwner && !isAssignedManager && !isOversight) {
      throw new ForbiddenException('You cannot view this request');
    }
  }

  private async log(
    requestId: string,
    action: string,
    performedBy: string,
    note?: string,
  ) {
    await this.audit.save(
      this.audit.create({ requestId, action, performedBy, note }),
    );
  }
}

import { BadRequestException, Injectable, NotFoundException, } from '@nestjs/common';
import { InjectRepository } from '@nestjs/typeorm';
import { FindOptionsWhere, ILike, Repository } from 'typeorm';
import { CreateResourceDto } from './dto/create-resource.dto';
import { UpdateResourceDto } from './dto/update-resource.dto';
import { QueryResourceDto } from './dto/query-resource.dto';
import { Resources } from './entities/resources.entity';
import { ResourceType } from './enum/resource-type.enum';

const INSOURCED_FIELDS = ['description', 'department'] as const;
const OUTSOURCED_FIELDS = ['price', 'label', 'batchNumber', 'vendor'] as const;

@Injectable()
export class ResourcesService {
  constructor(
    @InjectRepository(Resources)
    private readonly resourcesRepository: Repository<Resources>,
  ) {}

  async create(dto: CreateResourceDto): Promise<Resources>{
    const resource = dto.type === ResourceType.INSOURCED ? this.resourcesRepository.create({
        name: dto.name,
        type: dto.type,
        quantity: dto.quantity,
        description: dto.description,
        department: dto.department,
    })
    : this.resourcesRepository.create({
        name: dto.name,
        type: dto.type,
        price: dto.price,
        label: dto.label,
        quantity: dto.quantity,
        batchNumber: dto.batchNumber,
        vendor: dto.vendor,
    });
    return this.resourcesRepository.save(resource);
}
async findAll(query: QueryResourceDto): Promise<Resources[]>{
    const where: FindOptionsWhere<Resources> = {};
    if (query.name) where.name = ILike(`%${query.name}%`);
    if (query.type) where.type = query.type;

    return this.resourcesRepository.find({ where, order: { createdAt: 'DESC'} });

}
async findOne(id: string): Promise<Resources>
{
    const resource = await this.resourcesRepository.findOne({ where: { id }});
    if(!resource){
        throw new NotFoundException('Resource with id "${id}" not found');
    }
    return resource;
}
async update(id: string, dto: UpdateResourceDto): Promise<Resources>{
    const resource = await this.findOne(id);

    const forbidden = resource.type === ResourceType.INSOURCED ? OUTSOURCED_FIELDS : INSOURCED_FIELDS;
    const invalid = forbidden.filter(
        (field) => (dto as Record<string, unknown>)[field] !== undefined,);
        if(invalid.length){
            throw new BadRequestException(
                `'field(s)' [${invalid.join(',')}]) are not valid for ${resource.type} resources`,
            );
        }
        Object.assign(resource, dto);
        return this.resourcesRepository.save(resource);

}
}

import {
  ForbiddenException,
  Injectable,
  NotFoundException,
} from '@nestjs/common';
import { InjectRepository } from '@nestjs/typeorm';
import { Repository } from 'typeorm';
import { Team } from './entities/team.entity';
import { CreateTeamDto } from './dto/create-team.dto';
import { UpdateTeamDto } from './dto/update-team.dto';
import { Department } from '../departments/entities/department.entity';
import { User } from '../users/user.entity';
import { Project } from '../projects/entities/project.entity';
import { ResourceRequest } from '../resourcerequest/entities/resourcerequest.entity';
import { AuthTokenPayload } from '../auth/interfaces/auth-token-payload.interface';
import { UserRole } from '../users/user-role.enum';

@Injectable()
export class TeamsService {
  constructor(
    @InjectRepository(Team)
    private readonly teamRepository: Repository<Team>,
    @InjectRepository(Department)
    private readonly departmentRepository: Repository<Department>,
    @InjectRepository(User)
    private readonly userRepository: Repository<User>,
    @InjectRepository(Project)
    private readonly projectRepository: Repository<Project>,
    @InjectRepository(ResourceRequest)
    private readonly resourceRequestRepository: Repository<ResourceRequest>,
  ) {}

  async create(createTeamDto: CreateTeamDto) {
    const { departmentId, ...teamData } = createTeamDto;
    const department = await this.departmentRepository.findOneBy({ id: departmentId });
    if (!department) throw new NotFoundException(`Department with id ${departmentId} not found`);
    const team = this.teamRepository.create({ ...teamData, department });
    return this.teamRepository.save(team);
  }

  async findAll() {
    return this.teamRepository.find({ relations: { department: true } });
  }

  async findOne(id: number) {
    const team = await this.teamRepository.findOne({
      where: { id },
      relations: { department: true },
    });

    if (!team) {
      throw new NotFoundException(`Team with id ${id} not found`);
    }

    return team;
  }

  async findUsers(id: number) {
    await this.findOne(id);
    return this.userRepository.find({ where: { team: { id } } });
  }

  async findProjects(id: number) {
    await this.findOne(id);
    return this.projectRepository.find({
      where: { teams: { id } },
      relations: { teams: true },
    });
  }

  async findResourceRequests(id: number, user: AuthTokenPayload) {
    if (
      user.role !== UserRole.TEAM_LEAD &&
      user.role !== UserRole.TEAM_MEMBER
    ) {
      throw new ForbiddenException('Only team members can view team requests');
    }

    await this.findOne(id);
    const account = await this.userRepository.findOne({
      where: { id: user.sub },
      relations: { team: true },
    });
    if (!account?.team) {
      throw new ForbiddenException('Your account is not assigned to a team');
    }
    if (account.team.id !== id) {
      throw new NotFoundException(`Team with id ${id} not found`);
    }

    return this.resourceRequestRepository.find({
      where: { teamId: String(id) },
      order: { createdAt: 'DESC' },
    });
  }

  async update(id: number, updateTeamDto: UpdateTeamDto) {
    const team = await this.findOne(id);
    const { departmentId, ...teamData } = updateTeamDto;
    Object.assign(team, teamData);
    if (departmentId !== undefined) {
      const department = await this.departmentRepository.findOneBy({ id: departmentId });
      if (!department) throw new NotFoundException(`Department with id ${departmentId} not found`);
      team.department = department;
    }
    return this.teamRepository.save(team);
  }

  async remove(id: number) {
    const team = await this.findOne(id);
    await this.teamRepository.remove(team);
    return team;
  }
}

import { BadRequestException, ConflictException, Injectable, NotFoundException, UnauthorizedException } from '@nestjs/common';
import { InjectRepository } from '@nestjs/typeorm';
import { QueryFailedError, Repository } from 'typeorm';
import * as bcrypt from 'bcrypt';
import { UserRole } from './user-role.enum';
import { User } from './user.entity';
import { RegisterUserDto } from './dto/register-user.dto';
import { ChangePasswordDto } from './dto/change-password.dto';
import { UpdateMyProfileDto } from './dto/update-my-profile.dto';

const BCRYPT_ROUNDS = 12;

@Injectable()
export class UsersService {
  constructor(
    @InjectRepository(User)
    private readonly usersRepository: Repository<User>,
  ) {}

  async register(registerDto: RegisterUserDto) {
    const email = registerDto.email.trim().toLowerCase();
    const existingUser = await this.usersRepository.findOneBy({ email });
    if (existingUser) {
      throw new ConflictException('An account with this email already exists');
    }

    try {
      const user = await this.usersRepository.save(this.usersRepository.create({
        fullName: `${registerDto.firstName} ${registerDto.lastName}`.trim(),
        email,
        passwordHash: await bcrypt.hash(registerDto.password, BCRYPT_ROUNDS),
        role: null,
      }));
      return { user: { id: user.id, fullName: user.fullName, email: user.email, role: user.role } };
    } catch (error) {
      if (
        error instanceof QueryFailedError &&
        typeof error.driverError === 'object' && error.driverError !== null &&
        'code' in error.driverError && error.driverError.code === 'ER_DUP_ENTRY'
      ) {
        throw new ConflictException('An account with this email already exists');
      }
      throw error;
    }
  }

  async getMyProfile(userId: string){
    const user = await this .usersRepository.findOne({where: {id: userId},});
    
    if(!user){
      throw new NotFoundException('User Not found');
    }

    return {
      id: user.id,
      fullName: user.fullName,
      email: user.email,
      role: user.role,
      phoneNumber: user.phoneNumber,
      address: user.address,
      dateOfBirth: user.dateOfBirth,
      preferredPronoun: user.preferredPronoun
    };
  }

  async updateMyProfile(userId: string, dto: UpdateMyProfileDto) {
    if (Object.keys(dto).length === 0) {
      throw new BadRequestException('Provide at least one profile field to update');
    }
    if (dto.dateOfBirth && new Date(`${dto.dateOfBirth}T00:00:00Z`) > new Date()) {
      throw new BadRequestException('Date of birth cannot be in the future');
    }

    const user = await this.usersRepository.findOneBy({ id: userId });
    if (!user) {
      throw new NotFoundException('User not found');
    }

    Object.assign(user, dto);
    await this.usersRepository.save(user);
    return {
      id: user.id,
      fullName: user.fullName,
      email: user.email,
      role: user.role,
      phoneNumber: user.phoneNumber,
      address: user.address,
      dateOfBirth: user.dateOfBirth,
      preferredPronoun: user.preferredPronoun,
    };
  }

  async changeMyPassword(userId: string, changePasswordDto: ChangePasswordDto) {
    const user = await this.usersRepository
      .createQueryBuilder('user')
      .addSelect('user.passwordHash')
      .where('user.id = :userId', { userId })
      .getOne();

    if (!user) {
      throw new NotFoundException('User not found');
    }

    const currentPasswordMatches = await bcrypt.compare(
      changePasswordDto.currentPassword,
      user.passwordHash,
    );
    if (!currentPasswordMatches) {
      throw new UnauthorizedException('Current password is incorrect');
    }

    if (changePasswordDto.currentPassword === changePasswordDto.newPassword) {
      throw new BadRequestException('New password must be different from the current password');
    }

    user.passwordHash = await bcrypt.hash(
      changePasswordDto.newPassword,
      BCRYPT_ROUNDS,
    );
    await this.usersRepository.save(user);

    return { message: 'Password changed successfully' };
  }

  async deactivateMyAccount(userId: string){
    const user = await this.usersRepository.findOne({where: {id: userId},});
    if(!user){
      throw new NotFoundException('User Not Found');
    }

    if(!user.isActive){
      throw new BadRequestException('User already deactivated');
    }

    user.isActive = false;

    await this.usersRepository.save(user);

    return 'Your account has been deactivated successfully';
  }

  async reActivateMyAccount(userId: string){
    const user = await this.usersRepository.findOne({where: {id: userId},});

    if(!user){
      throw new NotFoundException('User not found')
    }

    if(user.isActive){
      throw new BadRequestException('account is already active');
    }
    
    user.isActive = true;

    await this.usersRepository.save(user);

    return 'Account successfully re-activated';
  }

  async assignRole(
    userId: string,
    role: UserRole,
    assignerId: string,
  ): Promise<Pick<User, 'id' | 'fullName' | 'email' | 'role'>> {
    const assigner = await this.usersRepository.findOneBy({ id: assignerId });
    if (!assigner) throw new NotFoundException('Assigning user not found');

    if (role === UserRole.MANAGING_DIRECTOR) {
      throw new BadRequestException(
        'The managing director role cannot be assigned to another user',
      );
    }

    if (role === UserRole.DEPARTMENT_HEAD && assigner.role !== UserRole.MANAGING_DIRECTOR) {
      throw new BadRequestException(
        'Only the managing director can assign the department head role',
      );
    }
    if (role === UserRole.TEAM_LEAD && assigner.role !== UserRole.DEPARTMENT_HEAD) {
      throw new BadRequestException(
        'Only department heads can assign the team lead role',
      );
    }
    if (role === UserRole.LINE_MANAGER && assigner.role !== UserRole.MANAGING_DIRECTOR) {
      throw new BadRequestException('Only the managing director can assign this role');
    }

    const user = await this.usersRepository.findOneBy({ id: userId });
    if (!user) {
      throw new NotFoundException('User not found');
    }
    if (user.role === UserRole.MANAGING_DIRECTOR) {
      throw new BadRequestException(
        'The managing director role cannot be changed through this endpoint',
      );
    }

    user.role = role;
    await this.usersRepository.save(user);

    return {
      id: user.id,
      fullName: user.fullName,
      email: user.email,
      role: user.role,
    };
  }

}
